### Tech Stack Requirements

[Source: architecture/tech-stack.md#core-languages-runtime]

**Runtime and Language:**
- Bun 1.1.x for JavaScript/TypeScript runtime
- TypeScript 5.9+ for type-safe development
- Use Bun native methods for file operations and process spawning

**Git Integration:**
- Use `Bun.spawn(['git', ...])` for Git operations
- No external Git libraries - leverage system Git installation
- Validate Git availability at startup

**Git Authentication:**
- **HTTPS Authentication**: Use `BMAD_GIT_TOKEN` environment variable
  - Token passed via Git credential helper or URL embedding
  - Example: `https://${Bun.env.BMAD_GIT_TOKEN}@github.com/user/repo.git`
  - Handle 401/403 errors with clear "authentication failed" message
- **SSH Authentication**: Use `BMAD_GIT_SSH_KEY` environment variable
  - Point to SSH private key file path
  - Git uses SSH agent or key file automatically
  - Example: `GIT_SSH_COMMAND='ssh -i ${Bun.env.BMAD_GIT_SSH_KEY}'`
  - Handle SSH errors with "SSH authentication failed" message
- **No Credentials**: Attempt anonymous clone first, fail gracefully if auth required
- **Error Handling**: Distinguish between network errors, auth errors, and repository not found

**Version Management:**
- semver 7.6.x for semantic version parsing and comparison
- @types/semver for TypeScript definitions
- Use semver.satisfies() for constraint matching

**Schema Validation:**
- Ajv 8.12.x for JSON schema validation (already in use from Story 3.1)
- JSON Schema Draft-07 for manifest schema

**File System:**
- Bun.file() for reading template files
- Bun.write() for writing cache and index files
- Template cache location: `~/.bmad/cache/templates/`
- Installed templates location: `~/.bmad/templates/{name}/{version}/`

**Git Authentication Error Messages:**

Provide clear, actionable error messages for authentication failures:

```typescript
// HTTPS Authentication Failure
class GitAuthenticationError extends Error {
  constructor(message: string, public readonly authType: 'https' | 'ssh') {
    super(message);
    this.name = 'GitAuthenticationError';
  }
}

// HTTPS token authentication failure
throw new GitAuthenticationError(
  'Failed to authenticate with Git repository using HTTPS.\n' +
  'Please ensure the BMAD_GIT_TOKEN environment variable is set with a valid personal access token.\n' +
  'See: https://docs.github.com/en/authentication/keeping-your-account-and-data-secure/creating-a-personal-access-token',
  'https'
);

// SSH authentication failure
throw new GitAuthenticationError(
  'Failed to authenticate with Git repository using SSH.\n' +
  'Please ensure:\n' +
  '  1. The BMAD_GIT_SSH_KEY environment variable points to your private key, or\n' +
  '  2. Your SSH agent is running with the correct key loaded\n' +
  'Test your SSH connection: ssh -T git@github.com',
  'ssh'
);

// Network/Connection error (not auth)
throw new Error(
  'Failed to connect to Git repository.\n' +
  'Please check:\n' +
  '  1. Your internet connection\n' +
  '  2. The repository URL is correct\n' +
  '  3. The repository exists and is accessible'
);

// Repository not found (could be auth or doesn't exist)
throw new Error(
  'Git repository not found.\n' +
  'This could mean:\n' +
  '  1. The repository does not exist\n' +
  '  2. You do not have access to this repository\n' +
  '  3. Authentication credentials are missing or invalid\n' +
  'Verify the repository URL and your access permissions.'
);
```

**Manifest Validation:**

```typescript
import Ajv from 'ajv';
import { createLogger } from '@checklist/core/utils/logger';

const logger = createLogger('checklist:templates:manifest');

// JSON Schema for TemplateManifest
export const manifestSchema = {
  $schema: 'http://json-schema.org/draft-07/schema#',
  type: 'object',
  required: ['name', 'version', 'description', 'author', 'repository', 'tags', 'license', 'files'],
  properties: {
    name: {
      type: 'string',
      pattern: '^[a-z0-9-]+$',
      minLength: 1,
      maxLength: 50,
    },
    version: {
      type: 'string',
      pattern: '^\\d+\\.\\d+\\.\\d+(-[a-zA-Z0-9.]+)?$', // Semver
    },
    description: {
      type: 'string',
      minLength: 10,
      maxLength: 200,
    },
    author: {
      oneOf: [
        { type: 'string' },
        {
          type: 'object',
          required: ['name'],
          properties: {
            name: { type: 'string' },
            email: { type: 'string', format: 'email' },
            url: { type: 'string', format: 'uri' },
          },
        },
      ],
    },
    repository: {
      type: 'object',
      required: ['type', 'url'],
      properties: {
        type: { type: 'string', enum: ['git'] },
        url: { type: 'string', format: 'uri' },
        directory: { type: 'string' },
      },
    },
    dependencies: {
      type: 'array',
      items: {
        type: 'object',
        required: ['name', 'version'],
        properties: {
          name: { type: 'string' },
          version: { type: 'string' },
          optional: { type: 'boolean' },
        },
      },
    },
    tags: {
      type: 'array',
      items: { type: 'string' },
      minItems: 1,
      maxItems: 10,
    },
    license: { type: 'string' },
    files: {
      type: 'array',
      items: { type: 'string' },
      minItems: 1,
    },
  },
};

// Common SPDX license identifiers
const COMMON_SPDX_LICENSES = [
  'MIT', 'Apache-2.0', 'GPL-3.0', 'GPL-2.0', 'BSD-3-Clause', 'BSD-2-Clause',
  'ISC', 'LGPL-3.0', 'LGPL-2.1', 'MPL-2.0', 'AGPL-3.0', 'Unlicense',
  'CC0-1.0', 'CC-BY-4.0', 'EPL-2.0', 'UNLICENSED'
];

export function validateLicense(license: string): { valid: boolean; warning?: string } {
  // Check if it's a common SPDX identifier
  if (COMMON_SPDX_LICENSES.includes(license)) {
    return { valid: true };
  }

  // Check if it's a "SEE LICENSE IN <filename>" pattern
  if (/^SEE LICENSE IN .+/.test(license)) {
    return { valid: true };
  }

  // Unrecognized license - warn but don't fail
  logger.warn({
    msg: 'Unrecognized license identifier',
    license,
    hint: 'Consider using a valid SPDX license identifier'
  });

  return {
    valid: true,
    warning: `Unrecognized license: ${license}. Consider using a valid SPDX identifier.`
  };
}

export function validateManifest(manifest: unknown): manifest is TemplateManifest {
  const ajv = new Ajv();
  const validate = ajv.compile(manifestSchema);
  const valid = validate(manifest);

  if (!valid) {
    logger.error({
      msg: 'Manifest validation failed',
      errors: validate.errors,
    });
    return false;
  }

  // Additional license validation
  if (manifest && typeof manifest === 'object' && 'license' in manifest) {
    const licenseValidation = validateLicense((manifest as any).license);
    if (licenseValidation.warning) {
      logger.warn({
        msg: 'License validation warning',
        warning: licenseValidation.warning
      });
    }
  }

  return valid;
}
```

```typescript
import { createLogger } from '@checklist/core/utils/logger';
import type { TemplateManifest } from '../types';

const logger = createLogger('checklist:templates:git-repository');

interface GitRepositoryConfig {
  cloneTimeout?: number;        // Default: 60000ms (60 seconds)
  maxRepoSize?: number;          // Default: 104857600 (100MB)
  maxClonesPerMinute?: number;   // Default: 10
  maxCacheSize?: number;         // Default: 1073741824 (1GB)
}

export class GitTemplateRepository {
  private cacheDir = '~/.bmad/cache/templates/';
  private config: Required<GitRepositoryConfig>;
  private static cloneAttempts: Map<string, number[]> = new Map(); // Rate limiting tracker

  constructor(
    private repositoryUrl: string,
    config?: GitRepositoryConfig
  ) {
    this.config = {
      cloneTimeout: config?.cloneTimeout ?? 60000,
      maxRepoSize: config?.maxRepoSize ?? 104857600,
      maxClonesPerMinute: config?.maxClonesPerMinute ?? 10,
      maxCacheSize: config?.maxCacheSize ?? 1073741824,
    };
  }

  async clone(targetDir: string): Promise<void> {
    // Check rate limiting
    await this.checkRateLimit();

    // Check available disk space
    await this.checkDiskSpace();

    logger.info({
      msg: 'Cloning template repository',
      url: this.repositoryUrl,
      target: targetDir,
      timeout: this.config.cloneTimeout,
    });

    // Create AbortController for timeout
    const controller = new AbortController();
    const timeoutId = setTimeout(() => {
      controller.abort();
      logger.error({
        msg: 'Clone operation timed out',
        url: this.repositoryUrl,
        timeout: this.config.cloneTimeout,
      });
    }, this.config.cloneTimeout);

    try {
      // Use shallow clone for performance
      const proc = Bun.spawn(
        ['git', 'clone', '--depth=1', this.repositoryUrl, targetDir],
        {
          stdout: 'pipe',
          stderr: 'pipe',
          signal: controller.signal
        }
      );

      const exitCode = await proc.exited;

      if (exitCode !== 0) {
        const stderr = await new Response(proc.stderr).text();
        throw new Error(`Git clone failed: ${stderr}`);
      }

      // Record successful clone for rate limiting
      this.recordCloneAttempt();

      logger.info({
        msg: 'Repository cloned successfully',
        target: targetDir,
      });
    } catch (error) {
      if (controller.signal.aborted) {
        // Clean up partial clone on timeout
        await this.cleanupFailedClone(targetDir);
        throw new Error(`Clone operation timed out after ${this.config.cloneTimeout}ms`);
      }

      logger.error({
        msg: 'Failed to clone repository',
        url: this.repositoryUrl,
        error: (error as Error).message,
      });

      // Clean up partial clone on error
      await this.cleanupFailedClone(targetDir);
      throw error;
    } finally {
      clearTimeout(timeoutId);
    }
  }

  private async cleanupFailedClone(targetDir: string): Promise<void> {
    try {
      logger.info({
        msg: 'Cleaning up failed clone',
        path: targetDir,
      });

      // Check if directory exists before cleanup
      const dirExists = await Bun.file(`${targetDir}/.git/config`).exists();
      if (!dirExists) {
        logger.debug({ msg: 'Clone directory does not exist, no cleanup needed' });
        return;
      }

      // Remove directory recursively using Bun.spawn (rm -rf)
      const proc = Bun.spawn(['rm', '-rf', targetDir], {
        stdout: 'pipe',
        stderr: 'pipe',
      });

      const exitCode = await proc.exited;

      if (exitCode !== 0) {
        const stderr = await new Response(proc.stderr).text();
        logger.warn({
          msg: 'Failed to cleanup clone directory',
          path: targetDir,
          error: stderr,
        });
        // Don't throw - cleanup failure shouldn't block error propagation
      } else {
        logger.info({
          msg: 'Clone directory cleaned up successfully',
          path: targetDir,
        });
      }
    } catch (cleanupError) {
      logger.warn({
        msg: 'Exception during cleanup',
        path: targetDir,
        error: (cleanupError as Error).message,
      });
      // Don't throw - cleanup failure shouldn't block error propagation
    }
  }

  private async checkRateLimit(): Promise<void> {
    const now = Date.now();
    const oneMinuteAgo = now - 60000;
    const userId = 'default'; // Could be extended to track per-user

    // Get attempts for this user
    const attempts = GitTemplateRepository.cloneAttempts.get(userId) ?? [];

    // Filter to attempts within last minute
    const recentAttempts = attempts.filter(time => time > oneMinuteAgo);

    if (recentAttempts.length >= this.config.maxClonesPerMinute) {
      logger.error({
        msg: 'Rate limit exceeded',
        attempts: recentAttempts.length,
        limit: this.config.maxClonesPerMinute,
      });
      throw new Error(
        `Rate limit exceeded: ${this.config.maxClonesPerMinute} clones per minute. Please try again later.`
      );
    }
  }

  private recordCloneAttempt(): void {
    const now = Date.now();
    const userId = 'default';
    const attempts = GitTemplateRepository.cloneAttempts.get(userId) ?? [];
    attempts.push(now);

    // Keep only last minute of attempts
    const oneMinuteAgo = now - 60000;
    const recentAttempts = attempts.filter(time => time > oneMinuteAgo);
    GitTemplateRepository.cloneAttempts.set(userId, recentAttempts);
  }

  private async checkDiskSpace(): Promise<void> {
    // Check available disk space in cache directory
    // Note: Bun doesn't have native disk space API, use system command
    const proc = Bun.spawn(['df', '-k', this.cacheDir], {
      stdout: 'pipe'
    });

    const output = await new Response(proc.stdout).text();
    const lines = output.split('\n');

    if (lines.length > 1) {
      const parts = lines[1].split(/\s+/);
      const availableKB = parseInt(parts[3], 10);
      const availableBytes = availableKB * 1024;

      if (availableBytes < this.config.maxRepoSize) {
        logger.error({
          msg: 'Insufficient disk space',
          available: availableBytes,
          required: this.config.maxRepoSize,
        });
        throw new Error(
          `Insufficient disk space: ${availableBytes} bytes available, need at least ${this.config.maxRepoSize} bytes`
        );
      }
    }
  }

  async fetch(localPath: string): Promise<void> {
    logger.info({
      msg: 'Fetching repository updates',
      path: localPath,
    });

    try {
      const proc = Bun.spawn(['git', '-C', localPath, 'pull', '--ff-only'], {
        stdout: 'pipe',
        stderr: 'pipe',
      });

      const exitCode = await proc.exited;

      if (exitCode !== 0) {
        const stderr = await new Response(proc.stderr).text();
        throw new Error(`Git fetch failed: ${stderr}`);
      }

      logger.info({
        msg: 'Repository updated successfully',
        path: localPath,
      });
    } catch (error) {
      logger.error({
        msg: 'Failed to fetch repository updates',
        path: localPath,
        error: (error as Error).message,
      });
      throw error;
    }
  }

  async listVersions(): Promise<string[]> {
    const cacheKey = this.getRepoCacheKey();
    const cachedPath = `${this.cacheDir}${cacheKey}`;

    // Clone to cache if not exists
    if (!await Bun.file(cachedPath).exists()) {
      await this.clone(cachedPath);
    }

    // List Git tags
    const proc = Bun.spawn(['git', '-C', cachedPath, 'tag', '--list'], {
      stdout: 'pipe',
    });

    const output = await new Response(proc.stdout).text();
    const versions = output
      .split('\n')
      .filter((v) => v.length > 0)
      .sort()
      .reverse(); // Latest first

    return versions;
  }

  async validateStructure(path: string): Promise<boolean> {
    const manifestPath = `${path}/template.json`;
    const manifestExists = await Bun.file(manifestPath).exists();

    if (!manifestExists) {
      logger.error({
        msg: 'Repository missing template.json',
        path,
      });
      return false;
    }

    return true;
  }

  private getRepoCacheKey(): string {
    // Create cache key from repository URL
    return Buffer.from(this.repositoryUrl).toString('base64');
  }
}
```

```typescript
import { createLogger } from '@checklist/core/utils/logger';
import type { TemplateIndex, TemplateMetadata } from './types';

const logger = createLogger('checklist:templates:registry');

export class TemplateRegistry {
  private index: TemplateIndex | null = null;
  private indexCacheTTL = 3600000; // 1 hour
  private lastIndexLoad = 0;

  constructor(private indexUrl: string) {}

  async loadIndex(): Promise<TemplateIndex> {
    const now = Date.now();

    // Return cached index if still valid
    if (this.index && now - this.lastIndexLoad < this.indexCacheTTL) {
      logger.debug({
        msg: 'Using cached template index',
        age: now - this.lastIndexLoad,
      });
      return this.index;
    }

    logger.info({
      msg: 'Loading template index',
      url: this.indexUrl,
    });

    try {
      // Fetch index file (supports file:// and https://)
      const indexContent = await this.fetchIndexFile();
      const index: TemplateIndex = JSON.parse(indexContent);

      this.index = index;
      this.lastIndexLoad = now;

      logger.info({
        msg: 'Template index loaded',
        templateCount: index.templates.length,
        version: index.version,
      });

      return index;
    } catch (error) {
      logger.error({
        msg: 'Failed to load template index',
        url: this.indexUrl,
        error: (error as Error).message,
      });
      throw error;
    }
  }

  async searchTemplates(query: string): Promise<TemplateMetadata[]> {
    const index = await this.loadIndex();
    const lowerQuery = query.toLowerCase();

    return index.templates.filter((template) => {
      return (
        template.name.toLowerCase().includes(lowerQuery) ||
        template.description.toLowerCase().includes(lowerQuery) ||
        template.tags.some((tag) => tag.toLowerCase().includes(lowerQuery))
      );
    });
  }

  async filterByTags(tags: string[]): Promise<TemplateMetadata[]> {
    const index = await this.loadIndex();
    const lowerTags = tags.map((t) => t.toLowerCase());

    return index.templates.filter((template) => {
      return template.tags.some((tag) =>
        lowerTags.includes(tag.toLowerCase())
      );
    });
  }

  async refreshIndex(): Promise<void> {
    logger.info({ msg: 'Manually refreshing template index' });
    this.index = null;
    this.lastIndexLoad = 0;
    await this.loadIndex();
  }

  private async fetchIndexFile(): Promise<string> {
    if (this.indexUrl.startsWith('file://')) {
      const path = this.indexUrl.replace('file://', '');
      return await Bun.file(path).text();
    } else if (this.indexUrl.startsWith('https://')) {
      const response = await fetch(this.indexUrl);
      if (!response.ok) {
        throw new Error(`Failed to fetch index: ${response.statusText}`);
      }
      return await response.text();
    } else {
      throw new Error(`Unsupported index URL protocol: ${this.indexUrl}`);
    }
  }
}
```

```typescript
import semver from 'semver';
import { createLogger } from '@checklist/core/utils/logger';
import type { GitTemplateRepository } from './repositories/GitTemplateRepository';

const logger = createLogger('checklist:templates:version-manager');

export class TemplateVersionManager {
  constructor(private repository: GitTemplateRepository) {}

  async resolveVersion(constraint: string): Promise<string> {
    logger.info({
      msg: 'Resolving version constraint',
      constraint,
    });

    // Handle 'latest' keyword
    if (constraint === 'latest') {
      const versions = await this.repository.listVersions();
      return versions[0]; // Already sorted latest first
    }

    // Get available versions
    const versions = await this.repository.listVersions();

    // Find matching version using semver
    const resolved = semver.maxSatisfying(versions, constraint);

    if (!resolved) {
      logger.error({
        msg: 'No version matching constraint',
        constraint,
        availableVersions: versions,
      });
      throw new Error(
        `No version matching constraint: ${constraint}. Available: ${versions.join(', ')}`
      );
    }

    logger.info({
      msg: 'Version resolved',
      constraint,
      resolved,
    });

    return resolved;
  }

  async listVersions(): Promise<string[]> {
    return await this.repository.listVersions();
  }

  compareVersions(v1: string, v2: string): number {
    return semver.compare(v1, v2);
  }

  isVersionCompatible(version: string, constraint: string): boolean {
    return semver.satisfies(version, constraint);
  }
}
```

**Dependency Resolver Implementation:**

```typescript
import { createLogger } from '@checklist/core/utils/logger';
import type {
  TemplateManifest,
  DependencyTree,
  DependencyNode,
} from './types';

const logger = createLogger('checklist:templates:dependency-resolver');

export class TemplateDependencyResolver {
  private visited = new Set<string>();
  private visiting = new Set<string>();

  constructor(
    private registry: TemplateRegistry,
    private versionManager: TemplateVersionManager,
    private templateLoader: TemplateLoader
  ) {}

  async resolveDependencies(
    manifest: TemplateManifest
  ): Promise<DependencyTree> {
    logger.info({
      msg: 'Resolving template dependencies',
      template: manifest.name,
    });

    this.visited.clear();
    this.visiting.clear();

    const dependencies = new Map<string, DependencyNode>();

    // Build dependency graph using DFS
    await this.buildDependencyGraph(manifest, dependencies, 0);

    // Calculate installation order (topological sort)
    const installOrder = this.calculateInstallOrder(dependencies);

    logger.info({
      msg: 'Dependencies resolved',
      totalDependencies: dependencies.size,
      installOrder,
    });

    return {
      root: manifest,
      dependencies,
      installOrder,
    };
  }

  private async buildDependencyGraph(
    manifest: TemplateManifest,
    dependencies: Map<string, DependencyNode>,
    depth: number
  ): Promise<void> {
    const name = manifest.name;

    // Detect circular dependencies
    if (this.visiting.has(name)) {
      throw new Error(`Circular dependency detected: ${name}`);
    }

    // Skip if already visited
    if (this.visited.has(name)) {
      return;
    }

    this.visiting.add(name);

    // Add current template to dependency map
    const node: DependencyNode = {
      manifest,
      dependencies: (manifest.dependencies ?? []).map((d) => d.name),
      depth,
      optional: false,
    };

    dependencies.set(name, node);

    // Recursively process dependencies
    for (const dep of manifest.dependencies ?? []) {
      try {
        const depManifest = await this.loadDependencyManifest(
          dep.name,
          dep.version
        );
        await this.buildDependencyGraph(depManifest, dependencies, depth + 1);
      } catch (error) {
        if (dep.optional) {
          logger.warn({
            msg: 'Optional dependency not available',
            dependency: dep.name,
          });
        } else {
          throw error;
        }
      }
    }

    this.visiting.delete(name);
    this.visited.add(name);
  }

  private calculateInstallOrder(
    dependencies: Map<string, DependencyNode>
  ): string[] {
    // Topological sort: deepest dependencies first
    const sorted = Array.from(dependencies.entries()).sort(
      ([, a], [, b]) => b.depth - a.depth
    );

    return sorted.map(([name]) => name);
  }

  private async loadDependencyManifest(
    name: string,
    version: string
  ): Promise<TemplateManifest> {
    // Check local cache first
    const cachePath = `~/.bmad/templates/${name}/${version}/template.json`;
    const cacheFile = Bun.file(cachePath);

    if (await cacheFile.exists()) {
      logger.debug({
        msg: 'Loading manifest from cache',
        name,
        version,
        path: cachePath,
      });

      // Use injected TemplateLoader for consistent loading and validation
      return await this.templateLoader.loadManifest(cachePath);
    }

    // Not in cache, need to fetch from registry
    logger.info({
      msg: 'Manifest not in cache, resolving from registry',
      name,
      version,
    });

    // Use injected TemplateRegistry to find repository URL
    const index = await this.registry.loadIndex();
    const template = index.templates.find((t) => t.name === name);

    if (!template) {
      throw new Error(`Template not found in registry: ${name}`);
    }

    // Clone repository
    const gitRepo = new GitTemplateRepository(template.repository);
    const tempPath = `~/.bmad/cache/templates/${name}-${version}`;
    await gitRepo.clone(tempPath);

    // Load and validate manifest using injected TemplateLoader from Story 3.1
    const manifest = await this.templateLoader.loadManifest(`${tempPath}/template.json`);

    // Verify version matches
    if (manifest.version !== version) {
      throw new Error(
        `Version mismatch: expected ${version}, got ${manifest.version}`
      );
    }

    logger.info({
      msg: 'Manifest loaded and validated',
      name,
      version,
    });

    return manifest;
  }
}
```
