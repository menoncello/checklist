**ConditionalEvaluationEngine Implementation:**

```typescript
import { createLogger } from '@checklist/core/utils/logger';
import type { TemplateSandbox } from '@checklist/core/templates/TemplateSandbox';
import type { VariableStore } from '@checklist/core/variables/VariableStore';

const logger = createLogger('checklist:workflow:conditional');

export class ConditionalEvaluationEngine {
  private cache = new Map<string, ConditionEvaluationResult>();

  constructor(
    private sandbox: TemplateSandbox,
    private variableStore: VariableStore
  ) {}

  async evaluateStepCondition(
    step: Step,
    context: ConditionalEvaluationContext
  ): Promise<boolean> {
    // No condition = always visible
    if (!step.condition) {
      return true;
    }

    // Check cache
    const cacheKey = this.getCacheKey(step, context);
    const cached = this.cache.get(cacheKey);
    if (cached && this.isCacheValid(cached, context)) {
      logger.debug({
        msg: 'Condition cache hit',
        stepId: step.id,
        condition: step.condition,
        result: cached.result,
      });
      return cached.result;
    }

    // Evaluate condition using TemplateSandbox (Story 3.1 + 3.5)
    const startTime = performance.now();
    try {
      const variables = this.variableStore.getAll(context.stepId);

      // Use TemplateSandbox.evaluateCondition() - already supports AND/OR/NOT from Story 3.5
      const result = await this.sandbox.evaluateCondition(
        step.condition,
        variables,
        `step-${step.id}`
      );

      const evaluationTime = performance.now() - startTime;

      // Cache result
      const evaluationResult: ConditionEvaluationResult = {
        stepId: step.id,
        condition: step.condition,
        result,
        evaluationTime,
        variablesUsed: this.extractVariables(step.condition),
        cached: false,
      };
      this.cache.set(cacheKey, evaluationResult);

      // Update step with result
      step.conditionResult = result;
      step.conditionEvaluatedAt = context.timestamp;

      logger.info({
        msg: 'Condition evaluated',
        stepId: step.id,
        condition: step.condition,
        result,
        evaluationTime,
      });

      return result;
    } catch (error) {
      logger.error({
        msg: 'Condition evaluation failed',
        stepId: step.id,
        condition: step.condition,
        error: (error as Error).message,
      });
      // Fail-safe: show step on error
      return true;
    }
  }

  async evaluateGroupCondition(
    groupId: string,
    steps: Step[],
    context: ConditionalEvaluationContext
  ): Promise<boolean> {
    const groupSteps = steps.filter((s) => s.groupId === groupId);
    if (groupSteps.length === 0) return true;

    const groupCondition = groupSteps[0].groupCondition;
    if (!groupCondition) return true;

    // Evaluate group condition
    const variables = this.variableStore.getAll(context.stepId);
    const result = await this.sandbox.evaluateCondition(
      groupCondition,
      variables,
      `group-${groupId}`
    );

    logger.info({
      msg: 'Group condition evaluated',
      groupId,
      condition: groupCondition,
      result,
      affectedSteps: groupSteps.length,
    });

    return result;
  }

  invalidateCache(): void {
    this.cache.clear();
    logger.debug({ msg: 'Condition cache invalidated' });
  }

  private getCacheKey(step: Step, context: ConditionalEvaluationContext): string {
    const variables = this.variableStore.getAll(context.stepId);
    const variablesHash = this.hashVariables(variables);
    return `${step.id}:${variablesHash}`;
  }

  private isCacheValid(
    cached: ConditionEvaluationResult,
    context: ConditionalEvaluationContext
  ): boolean {
    // Cache valid if variables haven't changed
    const currentVariables = this.variableStore.getAll(context.stepId);
    const currentHash = this.hashVariables(currentVariables);
    const cachedHash = this.hashVariables(
      Object.fromEntries(cached.variablesUsed.map((v) => [v, currentVariables[v]]))
    );
    return currentHash === cachedHash;
  }

  private hashVariables(variables: Record<string, any>): string {
    return JSON.stringify(variables); // Simple hash for now
  }

  private extractVariables(condition: string): string[] {
    // Extract variable names from condition expression
    const varPattern = /\b([a-zA-Z_][a-zA-Z0-9_]*)\b/g;
    const matches = condition.match(varPattern) || [];
    return [...new Set(matches)].filter(
      (v) => !['true', 'false', 'null', 'undefined'].includes(v)
    );
  }
}
```

**WorkflowEngine Integration:**

```typescript
// Update packages/core/src/workflow/WorkflowEngine.ts

export class WorkflowEngine {
  constructor(
    // ... existing dependencies
    private conditionalEngine: ConditionalEvaluationEngine
  ) {}

  async evaluateAllConditions(): Promise<void> {
    const startTime = performance.now();
    const context: ConditionalEvaluationContext = {
      variables: this.variableStore.getAll(),
      workflowState: this.state,
      timestamp: Date.now(),
    };

    // Evaluate group conditions first
    const groups = this.getUniqueGroups();
    for (const group of groups) {
      const groupResult = await this.conditionalEngine.evaluateGroupCondition(
        group.id,
        this.state.steps,
        context
      );
      group.conditionResult = groupResult;
    }

    // Evaluate individual step conditions in parallel
    await Promise.all(
      this.state.steps.map(async (step) => {
        // If step is in a group with false condition, skip evaluation
        if (step.groupId) {
          const group = groups.find((g) => g.id === step.groupId);
          if (group && group.conditionResult === false) {
            step.conditionResult = false;
            return;
          }
        }

        // Evaluate step condition
        await this.conditionalEngine.evaluateStepCondition(step, context);
      })
    );

    const duration = performance.now() - startTime;
    logger.info({
      msg: 'All conditions evaluated',
      totalSteps: this.state.steps.length,
      visibleSteps: this.getVisibleSteps().length,
      duration,
    });

    // Performance warning
    if (duration > 100 && this.state.steps.length > 100) {
      logger.warn({
        msg: 'Condition evaluation exceeded 100ms threshold',
        duration,
        stepCount: this.state.steps.length,
      });
    }

    this.emit('conditions-evaluated', { duration, timestamp: Date.now() });
  }

  getVisibleSteps(): Step[] {
    return this.state.steps.filter((step) => step.conditionResult !== false);
  }

  async reevaluateConditions(): Promise<void> {
    logger.info({ msg: 'Manual condition re-evaluation triggered' });
    this.conditionalEngine.invalidateCache();
    await this.evaluateAllConditions();
  }
}
```

**ConditionDebugger Implementation:**

```typescript
export class ConditionDebugger {
  constructor(
    private conditionalEngine: ConditionalEvaluationEngine,
    private variableStore: VariableStore
  ) {}

  getConditionReport(step: Step): ConditionDebugReport {
    if (!step.condition) {
      return {
        stepId: step.id,
        stepTitle: step.title,
        condition: undefined,
        result: true,
        variablesUsed: {},
        evaluationTime: 0,
        reason: 'No condition - always visible',
      };
    }

    const variablesUsed = this.extractVariablesUsed(step.condition);
    const variableValues = Object.fromEntries(
      variablesUsed.map((v) => [v, this.variableStore.get(v)])
    );

    const reason = step.conditionResult
      ? 'Condition evaluated to true - step visible'
      : `Condition evaluated to false - step hidden. Variables: ${JSON.stringify(variableValues)}`;

    return {
      stepId: step.id,
      stepTitle: step.title,
      condition: step.condition,
      result: step.conditionResult ?? true,
      variablesUsed: variableValues,
      evaluationTime: step.conditionEvaluatedAt
        ? Date.now() - step.conditionEvaluatedAt
        : 0,
      reason,
    };
  }

  getAllConditionReports(steps: Step[]): ConditionDebugReport[] {
    return steps.map((step) => this.getConditionReport(step));
  }

  exportDebugReport(steps: Step[], filePath: string): void {
    const reports = this.getAllConditionReports(steps);
    const reportText = this.formatReportAsText(reports);
    Bun.write(filePath, reportText);
    logger.info({ msg: 'Condition debug report exported', filePath });
  }

  private formatReportAsText(reports: ConditionDebugReport[]): string {
    let text = '# Condition Debug Report\n\n';
    text += `Generated: ${new Date().toISOString()}\n\n`;

    for (const report of reports) {
      text += `## Step: ${report.stepTitle} (${report.stepId})\n`;
      text += `- Condition: ${report.condition ?? 'None'}\n`;
      text += `- Result: ${report.result ? '✅ Visible' : '❌ Hidden'}\n`;
      text += `- Variables Used: ${JSON.stringify(report.variablesUsed, null, 2)}\n`;
      text += `- Reason: ${report.reason}\n\n`;
    }

    return text;
  }

  private extractVariablesUsed(condition: string): string[] {
    const varPattern = /\b([a-zA-Z_][a-zA-Z0-9_]*)\b/g;
    const matches = condition.match(varPattern) || [];
    return [...new Set(matches)].filter(
      (v) => !['true', 'false', 'null', 'undefined'].includes(v)
    );
  }
}
```
