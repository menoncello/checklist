**Conditional Evaluator Implementation:**

```typescript
export class ConditionalEvaluator {
  private static readonly IF_PATTERN = /\{\{#if\s+([^}]+)\}\}(.*?)(?:\{\{else\s+if\s+([^}]+)\}\}(.*?))*(?:\{\{else\}\}(.*?))?\{\{\/if\}\}/gs;

  constructor(
    private variableStore: VariableStore,
    private sandbox: TemplateSandbox
  ) {}

  evaluate(template: string, context: EvaluationContext): ConditionalResult {
    const startTime = performance.now();
    let conditionsMet = 0;

    const output = template.replace(this.IF_PATTERN, (match, condition, ifContent, ...rest) => {
      // Parse condition safely using AST
      const conditionResult = this.evaluateCondition(condition, context);

      if (conditionResult) {
        conditionsMet++;
        return this.processContent(ifContent, context);
      }

      // Check else-if branches
      for (let i = 0; i < rest.length - 1; i += 2) {
        const elseIfCondition = rest[i];
        const elseIfContent = rest[i + 1];

        if (elseIfCondition && this.evaluateCondition(elseIfCondition, context)) {
          conditionsMet++;
          return this.processContent(elseIfContent, context);
        }
      }

      // Else branch
      const elseContent = rest[rest.length - 1];
      if (elseContent) {
        return this.processContent(elseContent, context);
      }

      return '';
    });

    const evaluationTime = performance.now() - startTime;

    return {
      output,
      conditionsMet,
      evaluationTime,
    };
  }

  private async evaluateCondition(condition: string, context: EvaluationContext): Promise<boolean> {
    // Use TemplateSandbox.evaluateCondition() for safe evaluation
    // TemplateSandbox already includes AST parsing and validation (Story 3.1)
    // Support operators: ==, !=, >, <, >=, <=, &&, ||, !
    // Example: "count > 10 && status == 'active'"

    try {
      // TemplateSandbox.evaluateCondition() handles sanitization and AST validation
      const result = await this.sandbox.evaluateCondition(
        condition,
        context.variables,
        'conditional-evaluator'
      );
      return result;
    } catch (error) {
      logger.error({
        msg: 'Condition evaluation failed',
        condition,
        error: (error as Error).message
      });
      return false; // Fail safe: false on error
    }
  }
}
```

**Loop Evaluator Implementation:**

```typescript
export class LoopEvaluator {
  private static readonly EACH_PATTERN = /\{\{#each\s+([^}]+)\}\}(.*?)\{\{\/each\}\}/gs;
  private readonly maxIterations = 10000;

  constructor(
    private variableStore: VariableStore,
    private sandbox: TemplateSandbox
  ) {}

  evaluate(template: string, context: EvaluationContext): LoopResult {
    const startTime = performance.now();
    let totalIterations = 0;
    let limitReached = false;

    const output = template.replace(this.EACH_PATTERN, (match, itemsVar, content) => {
      // Get items from variable store
      const items = this.variableStore.get(itemsVar, context.stepId);

      if (!Array.isArray(items)) {
        logger.warn({ msg: 'Loop items not array', itemsVar });
        return '';
      }

      if (items.length > this.maxIterations) {
        limitReached = true;
        logger.warn({
          msg: 'Loop iteration limit reached',
          itemsVar,
          itemCount: items.length,
          limit: this.maxIterations
        });
      }

      const iterationLimit = Math.min(items.length, this.maxIterations);
      const outputs: string[] = [];

      for (let i = 0; i < iterationLimit; i++) {
        const loopContext: LoopContext = {
          items,
          index: i,
          first: i === 0,
          last: i === items.length - 1,
          parent: context.loopContext,
        };

        // Create evaluation context with special loop variables
        const loopVars = {
          ...context.variables,
          this: items[i],
          '@index': i,
          '@first': loopContext.first,
          '@last': loopContext.last,
        };

        const loopEvalContext: EvaluationContext = {
          ...context,
          variables: loopVars,
          loopContext,
        };

        // Process content with loop variables
        outputs.push(this.processContent(content, loopEvalContext));
        totalIterations++;
      }

      return outputs.join('');
    });

    const evaluationTime = performance.now() - startTime;

    return {
      output,
      iterations: totalIterations,
      evaluationTime,
      limitReached,
    };
  }
}
```

```typescript
export class TemplateFunctions {
  private static readonly FUNCTION_PATTERN = /\$\{fn:([a-zA-Z0-9_]+)\(([^)]*)\)\}/g;
  private registry: FunctionRegistry;

  constructor() {
    this.registry = new FunctionRegistryImpl();
    this.registerBuiltInFunctions();
  }

  private registerBuiltInFunctions(): void {
    // String functions
    this.registry.register('uppercase', {
      name: 'uppercase',
      execute: ([str]) => String(str).toUpperCase(),
      validate: (args) => args.length === 1,
      safe: true,
    });

    this.registry.register('lowercase', {
      name: 'lowercase',
      execute: ([str]) => String(str).toLowerCase(),
      validate: (args) => args.length === 1,
      safe: true,
    });

    // Array functions
    this.registry.register('length', {
      name: 'length',
      execute: ([arr]) => Array.isArray(arr) ? arr.length : String(arr).length,
      validate: (args) => args.length === 1,
      safe: true,
    });

    this.registry.register('join', {
      name: 'join',
      execute: ([arr, separator = ',']) => Array.isArray(arr) ? arr.join(separator) : '',
      validate: (args) => args.length >= 1,
      safe: true,
    });

    // Utility functions
    this.registry.register('default', {
      name: 'default',
      execute: ([value, defaultValue]) => value ?? defaultValue,
      validate: (args) => args.length === 2,
      safe: true,
    });

    // Date functions
    this.registry.register('formatDate', {
      name: 'formatDate',
      execute: ([date, format]) => this.formatDateImpl(date, format),
      validate: (args) => args.length === 2,
      safe: true,
    });
  }

  evaluate(template: string, context: EvaluationContext): string {
    return template.replace(this.FUNCTION_PATTERN, (match, fnName, argsStr) => {
      const fn = this.registry.get(fnName);

      if (!fn) {
        logger.warn({ msg: 'Unknown template function', fnName });
        return match;
      }

      if (!fn.safe) {
        throw new SecurityError(`Function '${fnName}' is not sandbox-safe`);
      }

      // Parse arguments (may contain variable references)
      const args = this.parseArgs(argsStr, context);

      if (!fn.validate(args)) {
        logger.error({ msg: 'Invalid function arguments', fnName, args });
        return match;
      }

      try {
        return String(fn.execute(args, context));
      } catch (error) {
        logger.error({ msg: 'Function execution failed', fnName, error });
        return match;
      }
    });
  }
}
```

**Math Expression Evaluator Implementation:**

```typescript
export class MathExpressionEvaluator {
  private static readonly MATH_PATTERN = /\$\{([^}]+[+\-*/%][^}]+)\}/g;

  constructor(
    private sandbox: TemplateSandbox,
    private variableStore: VariableStore
  ) {}

  async evaluate(template: string, context: EvaluationContext): Promise<string> {
    // Process all math expressions in parallel
    const promises: Promise<string>[] = [];
    const matches: Array<{ match: string; expression: string }> = [];

    // Collect all matches first
    let match: RegExpExecArray | null;
    const pattern = new RegExp(this.MATH_PATTERN.source, this.MATH_PATTERN.flags);
    while ((match = pattern.exec(template)) !== null) {
      matches.push({ match: match[0], expression: match[1] });
    }

    // Evaluate all expressions
    for (const { match, expression } of matches) {
      promises.push(this.evaluateExpression(match, expression, context));
    }

    const results = await Promise.all(promises);

    // Replace matches with results
    let output = template;
    matches.forEach(({ match }, index) => {
      output = output.replace(match, results[index]);
    });

    return output;
  }

  private async evaluateExpression(
    match: string,
    expression: string,
    context: EvaluationContext
  ): Promise<string> {
    try {
      // Validate expression is mathematical (no dangerous operations)
      if (!this.isSafeMathExpression(expression)) {
        logger.warn({ msg: 'Unsafe math expression', expression });
        return match;
      }

      // Substitute variables in expression
      const substituted = this.substituteVariables(expression, context);

      // Use TemplateSandbox.executeExpression() for safe evaluation
      // This handles AST parsing, validation, and execution with resource limits
      const result = await this.sandbox.executeExpression(
        substituted,
        {},
        'math-evaluator',
        50 // 50ms timeout per expression
      );

      // Validate result is number
      if (typeof result !== 'number' || !isFinite(result)) {
        logger.warn({ msg: 'Invalid math result', expression, result });
        return match;
      }

      return String(result);
    } catch (error) {
      logger.error({
        msg: 'Math expression evaluation failed',
        expression,
        error: (error as Error).message
      });
      return match; // Return original on error (fail-safe)
    }
  }

  private isSafeMathExpression(expression: string): boolean {
    // Only allow: numbers, variables, math operators, parentheses, spaces
    const safePattern = /^[\d\w\s+\-*/%().]+$/;
    return safePattern.test(expression);
  }

  private substituteVariables(expression: string, context: EvaluationContext): string {
    // Replace variable names with their values
    const varPattern = /\b([a-zA-Z_][a-zA-Z0-9_]*)\b/g;

    return expression.replace(varPattern, (match, varName) => {
      // Skip if it's a math function
      if (['min', 'max', 'round', 'floor', 'ceil', 'abs'].includes(varName)) {
        return match;
      }

      const value = this.variableStore.get(varName, context.stepId);

      if (typeof value === 'number') {
        return String(value);
      }
```

**Template Engine Pipeline Integration:**

```typescript
// Update packages/core/src/templates/TemplateEngine.ts

export class TemplateEngine {
  constructor(
    private sandbox: TemplateSandbox,
    private variableStore: VariableStore,
    private substitutor: VariableSubstitutor,
    private mathEvaluator: MathExpressionEvaluator,
    private functionEvaluator: TemplateFunctions,
    private conditionalEvaluator: ConditionalEvaluator,
    private loopEvaluator: LoopEvaluator
  ) {}

  async processTemplate(template: string, stepId?: string): Promise<string> {
    const startTime = performance.now();

    // Create evaluation context
    const context: EvaluationContext = {
      variables: this.variableStore.getAll(stepId),
      stepId,
      nestingDepth: 0,
      maxNestingDepth: 10,
      performanceMetrics: {
        startTime,
        evaluations: 0,
        maxExecutionTime: 50,  // AC: 8 - <50ms for complex templates
      },
    };

    // Pipeline: Variable substitution → Math → Functions → Conditionals → Loops
    let output = template;

    // Step 1: Variable substitution (Story 3.4)
    const substResult = this.substitutor.substitute(output, stepId);
    output = substResult.output;

    // Step 2: Math expressions
    output = this.mathEvaluator.evaluate(output, context);

    // Step 3: Template functions
    output = this.functionEvaluator.evaluate(output, context);

    // Step 4: Conditionals (must be before loops to support if inside each)
    const conditionalResult = this.conditionalEvaluator.evaluate(output, context);
    output = conditionalResult.output;

    // Step 5: Loops
    const loopResult = this.loopEvaluator.evaluate(output, context);
    output = loopResult.output;

    // Check performance requirement
    const duration = performance.now() - startTime;

    if (duration > context.performanceMetrics.maxExecutionTime) {
      logger.warn({
        msg: 'Template processing exceeded performance target',
        duration,
        limit: context.performanceMetrics.maxExecutionTime,
      });
    }

    // Log performance metrics
    logger.info({
      msg: 'Template processed with advanced features',
      duration,
      variableCount: substResult.variablesUsed.length,
      conditionsMet: conditionalResult.conditionsMet,
      loopIterations: loopResult.iterations,
    });

    return output;
  }
}
```
